Hello, WBSKT

The same few lines in each: connect with a name and a policy PIN, send a reading, take a command.

#include <WiFi.h>
#include <Wbskt.h>

WbsktClient wbskt("greenhouse-sensor", "483920");

void setup() {
  WiFi.begin("home-wifi", "wifi-password");
  wbskt.onMessage([](const WbsktMessage& message) {
    if (strcmp(message.type, "heater") == 0) {
      switchHeater(message.payload["on"]);
    }
  });
  wbskt.begin();
}

void loop() {
  wbskt.loop();
  static unsigned long last = 0;
  if (millis() - last < 10000) return;
  last = millis();
  JsonDocument reading;
  reading["celsius"] = readTemperature();
  wbskt.send("temperature", reading);
}
using Wbskt.Client.Sdk;

await using var wbskt = new WbsktClient("greenhouse-sensor", "483920");

wbskt.OnMessageReceived += (type, payload, commandId) =>
{
    if (type == "heater") heater.Switch(payload);
};

await wbskt.StartAsync();
await wbskt.SendAsync("temperature", new { celsius = ReadTemperature() });
from wbskt import WbsktClient

client = WbsktClient("greenhouse-sensor", policy_pin="483920")

def handle(message):
    if message.type == "heater":
        heater.switch(message.payload["on"])

client.on_message = handle
client.start()
client.send("temperature", {"celsius": read_temperature()})

Another language?

Anything that speaks HTTPS and WebSocket can be a client. Follow the client protocol, then run the shared conformance tests against your client to check it behaves like the SDKs.
